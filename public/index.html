const express = require('express'); const session = require('express-session'); const MongoStore = require('connect-mongo'); const path = require('path'); const { connectDB, getDB } = require('./db'); const app = express(); const PORT = process.env.PORT || 3000; app.use(express.json()); app.use(express.urlencoded({ extended: true })); const sessionSecret = process.env.SESSION_SECRET || 'j2c_inovacoes_segredo_super_secreto_2026'; app.use(session({ secret: sessionSecret, resave: false, saveUninitialized: false, store: MongoStore.create({ mongoUrl: process.env.MONGODB_URI || 'mongodb://localhost:27017/alvara01', touchAfter: 24 * 3600 }), cookie: { maxAge: 1000 * 60 * 60 * 24 * 7, secure: false, httpOnly: true } })); app.use(express.static(path.join(__dirname, 'public'))); function requireAuth(req, res, next) { if (!req.session.userId) { return res.status(401).json({ erro: 'Deve inscrever-se ou iniciar sessão para aceder à informação.' }); } next(); } app.post('/api/register', async (req, res) => { try { const { nome, email, telefone, senha } = req.body; if (!nome || !email || !telefone || !senha) { return res.status(400).json({ erro: 'Preencha todos os campos obrigatórios.' }); } const db = getDB(); const usersCollection = db.collection('users'); const existingUser = await usersCollection.findOne({ email }); if (existingUser) { return res.status(400).json({ erro: 'Já existe uma conta registada com este email.' }); } const newUser = { nome, email, telefone, senha, createdAt: new Date() }; const result = await usersCollection.insertOne(newUser); req.session.userId = result.insertedId; req.session.userName = nome; res.json({ sucesso: true, mensagem: 'Inscrição efetuada com sucesso!', nome }); } catch (err) { console.error('Erro no registo:', err); res.status(500).json({ erro: 'Erro interno no servidor ao criar conta.' }); } }); app.post('/api/login', async (req, res) => { try { const { email, senha } = req.body; if (!email || !senha) { return res.status(400).json({ erro: 'Informe o email e a senha.' }); } const db = getDB(); const usersCollection = db.collection('users'); const user = await usersCollection.findOne({ email, senha }); if (!user) { return res.status(400).json({ erro: 'Credenciais inválidas. Verifique os dados.' }); } req.session.userId = user._id; req.session.userName = user.nome; res.json({ sucesso: true, mensagem: 'Sessão iniciada com sucesso!', nome: user.nome }); } catch (err) { console.error('Erro no login:', err); res.status(500).json({ erro: 'Erro interno ao iniciar sessão.' }); } }); app.post('/api/logout', (req, res) => { req.session.destroy(err => { if (err) return res.status(500).json({ erro: 'Erro ao terminar sessão.' }); res.clearCookie('connect.sid'); res.json({ sucesso: true, mensagem: 'Sessão encerrada.' }); }); }); app.get('/api/me', (req, res) => { if (!req.session.userId) { return res.status(401).json({ autenticado: false }); } res.json({ autenticado: true, nome: req.session.userName }); }); async function executarPesquisaGovernamentalAngola(termo) { const queryWeb = `${termo} alvará licença requisitos angola site:gov.ao`; const urlBusca = `https://html.duckduckgo.com/html/?q=${encodeURIComponent(queryWeb)}`; try { const response = await fetch(urlBusca, { headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) J2C-Inovacoes-Bot/3.0' }, signal: AbortSignal.timeout(9500) }); if (!response.ok) throw new Error('Erro na resposta da pesquisa web institucional.'); const html = await response.text(); return { fonte: "Pesquisa Web Direta em Portais do Governo de Angola (.gov.ao)", sucessoWeb: true, brutoHtml: html }; } catch (error) { console.error('Timeout ou falha na web:', error); return { fonte: "Base de Conhecimento Interna J2C", sucessoWeb: false }; } } app.get('/api/search', requireAuth, async (req, res) => { try { const query = req.query.q; if (!query) { return res.status(400).json({ erro: 'Insira uma actividade económica para pesquisar.' }); } const db = getDB(); await db.collection('history').insertOne({ userId: req.session.userId, query: query, createdAt: new Date() }); await executarPesquisaGovernamentalAngola(query); const knowledgeCollection = db.collection('knowledge'); let registro = await knowledgeCollection.findOne({ atividade: { $regex: new RegExp(query, 'i') } }); let resultadoFinal; if (registro) { resultadoFinal = registro; } else { resultadoFinal = { atividade: query, tipoAlvará: `Alvará Comercial / Licença de Exploração para o sector de ${query}`, orgaoEmissor: "Governo Provincial / Administração Municipal / Ministério da Tutela em Angola", requisitos: [ "Cópia do Bilhete de Identidade (BI) dos titulares ou sócios", "NIF (Número de Identificação Fiscal) da empresa emitido pela AGT", "Certidão Comercial / Registo Comercial emitido pelo GUE (Guiché Único de Empresa)", "Croquis de localização e contrato de arrendamento ou título de propriedade do espaço", "Parecer técnico de salubridade, segurança contra incêndios ou vistoria específica do sector" ], prazoEmissao: "15 a 30 dias úteis após submissão regular do expediente", contacto: "Central de Atendimento do SIAC / Repartição Fiscal da AGT / Administração Local", localizacion: "Balcões de Atendimento Integrado (SIAC) ou sede da Administração Municipal em Angola" }; } res.json({ sucesso: true, resultado: resultadoFinal }); } catch (err) { console.error('Erro na rota de pesquisa:', err); res.status(500).json({ erro: 'Ocorreu um erro ao processar a pesquisa nos portais institucionais.' }); } }); app.get('/api/history', requireAuth, async (req, res) => { try { const db = getDB(); const history = await db.collection('history') .find({ userId: req.session.userId }) .sort({ createdAt: -1 }) .limit(10) .toArray(); res.json({ sucesso: true, history }); } catch (err) { console.error('Erro ao buscar histórico:', err); res.status(500).json({ erro: 'Erro ao carregar o histórico de pesquisas.' }); } }); connectDB((err) => { if (err) { console.error('Erro fatal ao conectar à Base de Dados:', err); process.exit(1); } app.listen(PORT, '0.0.0.0', () => { console.log(`Plataforma J2C Inovações a operar na porta ${PORT} focada em Angola 🇦🇴`); }); });
